// 2. Procesa el DNI
function procesarDni(dni) {
  Logger.log('=== NUEVA VERSION - procesarDni: ' + dni + ' ===');
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hojaSocios = ss.getSheetByName("Socios"); 
  
  var hojaAsistencias = ss.getSheetByName("Asistencias");
  if (!hojaAsistencias) {
    hojaAsistencias = ss.insertSheet("Asistencias");
    hojaAsistencias.appendRow(["Fecha y Hora", "DNI", "Socio", "Estado en Sistema"]);
  }
  
  if (!dni) return { estatus: "error", mensaje: "Por favor, ingresa un DNI." };
  
  var datosSocios = hojaSocios.getDataRange().getValues();
  
  var colDni = 0;         
  var colNombre = 1;      
  var colApellido = 2;    
  var colVencimiento = 3; 
  
  var socioEncontrado = false;
  var nombreCompleto = "";
  var fechaVencimiento = null;
  
  for (var i = 1; i < datosSocios.length; i++) {
    var dniCelda = datosSocios[i][colDni] ? datosSocios[i][colDni].toString().trim() : "";
    if (dniCelda === dni.trim()) {
      socioEncontrado = true;
      nombreCompleto = datosSocios[i][colNombre] + " " + datosSocios[i][colApellido];
      fechaVencimiento = datosSocios[i][colVencimiento];
      break;
    }
  }
  
  if (socioEncontrado) {
    var hoy = new Date();
    hoy.setHours(0,0,0,0);
    var estadoCuota = "AL DIA";
    var textoVencimiento = "Sin fecha";
    
    if (fechaVencimiento) {
      var fechaVenDate = new Date(fechaVencimiento);
      fechaVenDate.setMinutes(fechaVenDate.getMinutes() + fechaVenDate.getTimezoneOffset());
      
      var dia = String(fechaVenDate.getDate()).padStart(2, '0');
      var mes = String(fechaVenDate.getMonth() + 1).padStart(2, '0');
      var anio = fechaVenDate.getFullYear();
      textoVencimiento = dia + "/" + mes + "/" + anio;
      
      fechaVenDate.setHours(0,0,0,0);
      if (hoy > fechaVenDate) {
        estadoCuota = "VENCIDO";
      }
    }
    
    hojaAsistencias.appendRow([new Date(), dni, nombreCompleto, estadoCuota]);

    // Sync directo a Firestore (igual que debugSync, que sabemos que funciona)
    try {
      var fbToken = getFirebaseToken_();
      if (fbToken) {
        var fbResp = UrlFetchApp.fetch(
          'https://firestore.googleapis.com/v1/projects/new-life-app-f951c/databases/(default)/documents/ingresosActivos',
          {
            method: 'POST',
            contentType: 'application/json',
            headers: { Authorization: 'Bearer ' + fbToken },
            payload: JSON.stringify({
              fields: {
                dni:       { stringValue: String(dni).trim()      },
                nombre:    { stringValue: String(nombreCompleto)  },
                estado:    { stringValue: String(estadoCuota)     },
                fechaHora: { timestampValue: new Date().toISOString() },
                activo:    { booleanValue: true }
              }
            }),
            muteHttpExceptions: true
          }
        );
        Logger.log('Firestore ingreso HTTP ' + fbResp.getResponseCode() + ' - ' + dni);
      }
    } catch(e) { Logger.log('Firestore sync error: ' + e.message); }

    if (estadoCuota === "AL DIA") {
      return { 
        estatus: "aldia", 
        mensaje: "¡BIENVENIDO/A, " + nombreCompleto.toUpperCase() + "!", 
        subtexto: "Acceso Concedido • Tu cuota vence el: " + textoVencimiento 
      };
    } else {
      return { 
        estatus: "vencido", 
        mensaje: "CUOTA VENCIDA", 
        subtexto: nombreCompleto.toUpperCase() + " (Venció el " + textoVencimiento + ") • Pasa por recepción" 
      };
    }
  } else {
    return { estatus: "noencontrado", mensaje: "DNI NO ENCONTRADO", subtexto: "Por favor, regístrate en recepción." };
  }
}

function getFirebaseToken_() {
  var cache = CacheService.getScriptCache();
  var cached = cache.get('fb_token');
  if (cached) return cached;

  var sa = JSON.parse(
    PropertiesService.getScriptProperties().getProperty('FIREBASE_SA')
  );

  var header = Utilities.base64EncodeWebSafe(
    JSON.stringify({ alg: 'RS256', typ: 'JWT' })
  ).replace(/=+$/, '');

  var now = Math.floor(Date.now() / 1000);
  var claim = Utilities.base64EncodeWebSafe(
    JSON.stringify({
      iss:   sa.client_email,
      scope: 'https://www.googleapis.com/auth/datastore',
      aud:   'https://oauth2.googleapis.com/token',
      iat:   now,
      exp:   now + 3600
    })
  ).replace(/=+$/, '');

  var toSign = header + '.' + claim;
  var sig = Utilities.base64EncodeWebSafe(
    Utilities.computeRsaSha256Signature(toSign, sa.private_key)
  ).replace(/=+$/, '');

  var resp = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', {
    method: 'POST',
    contentType: 'application/x-www-form-urlencoded',
    payload: 'grant_type=urn%3Aietf%3Aparams%3Aoauth%3Agrant-type%3Ajwt-bearer&assertion=' + toSign + '.' + sig,
    muteHttpExceptions: true
  });

  var token = JSON.parse(resp.getContentText()).access_token;
  if (token) cache.put('fb_token', token, 3500);
  return token;
}

function isAlreadyActive_(dni, token) {
  var cutoff = new Date(Date.now() - 90 * 60 * 1000).toISOString();
  var url = 'https://firestore.googleapis.com/v1/projects/new-life-app-f951c'
          + '/databases/(default)/documents:runQuery';
  var resp = UrlFetchApp.fetch(url, {
    method: 'POST',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({
      structuredQuery: {
        from: [{ collectionId: 'ingresosActivos' }],
        where: {
          compositeFilter: {
            op: 'AND',
            filters: [
              { fieldFilter: { field: { fieldPath: 'dni' }, op: 'EQUAL',        value: { stringValue:   String(dni) } } },
              { fieldFilter: { field: { fieldPath: 'fechaHora' }, op: 'GREATER_THAN', value: { timestampValue: cutoff    } } }
            ]
          }
        },
        limit: 1
      }
    }),
    muteHttpExceptions: true
  });
  if (resp.getResponseCode() !== 200) return false;
  var result = JSON.parse(resp.getContentText());
  return result.length > 0 && !!result[0].document;
}

function syncIngresoFirestore_(dni, nombre, estado, fechaVencimientoISO) {
  try {
    var token = getFirebaseToken_();

    // Evitar duplicados: si la persona ya tiene ingreso activo en los últimos 75 min, no registrar
    if (isAlreadyActive_(dni, token)) return;

    var base = 'https://firestore.googleapis.com/v1/projects/new-life-app-f951c'
             + '/databases/(default)/documents';
    var nowISO = new Date().toISOString();

    // 1. Registrar el ingreso en ingresosActivos (historial)
    UrlFetchApp.fetch(base + '/ingresosActivos', {
      method: 'POST',
      contentType: 'application/json',
      headers: { Authorization: 'Bearer ' + token },
      payload: JSON.stringify({
        fields: {
          dni:       { stringValue: String(dni)    },
          nombre:    { stringValue: String(nombre) },
          estado:    { stringValue: String(estado) },
          fechaHora: { timestampValue: nowISO },
          activo:    { booleanValue: true }
        }
      }),
      muteHttpExceptions: true
    });

    // 2. Actualizar/crear el documento del socio con su estado de cuota actual
    var fechaVenField = fechaVencimientoISO
      ? { timestampValue: fechaVencimientoISO }
      : { nullValue: null };

    UrlFetchApp.fetch(base + '/socios/' + String(dni).trim(), {
      method: 'PATCH',
      contentType: 'application/json',
      headers: { Authorization: 'Bearer ' + token },
      payload: JSON.stringify({
        fields: {
          dni:              { stringValue: String(dni)    },
          nombre:           { stringValue: String(nombre) },
          estado:           { stringValue: String(estado) },
          fechaVencimiento: fechaVenField,
          ultimaVisita:     { timestampValue: nowISO }
        }
      }),
      muteHttpExceptions: true
    });

  } catch (e) {
    console.error('Firestore sync error: ' + e.message);
  }
}

// ─── SINCRONIZACIÓN MASIVA: corre una sola vez desde el editor ───────────────
// Lee TODOS los socios del Excel y los sube a Firestore colección "socios".
// Después de correrlo, la app muestra el estado de cuota de cada uno.
function syncTodosLosSocios() {
  var ss        = SpreadsheetApp.getActiveSpreadsheet();
  var hoja      = ss.getSheetByName('Socios');
  var datos     = hoja.getDataRange().getValues();
  var token     = getFirebaseToken_();
  var projectId = 'new-life-app-f951c';
  var dbPath    = 'projects/' + projectId + '/databases/(default)/documents';
  var commitUrl = 'https://firestore.googleapis.com/v1/' + dbPath + ':commit';

  var hoy = new Date();
  hoy.setHours(0, 0, 0, 0);

  var writes = [];
  var total  = 0;

  for (var i = 1; i < datos.length; i++) {
    var dni      = datos[i][0] ? String(datos[i][0]).trim() : '';
    var nombre   = datos[i][1] ? String(datos[i][1]).trim() : '';
    var apellido = datos[i][2] ? String(datos[i][2]).trim() : '';
    var fechaVen = datos[i][3];
    if (!dni) continue;

    var nombreCompleto = (nombre + ' ' + apellido).trim();
    var estado         = 'AL DIA';
    var fechaVenField  = { nullValue: null };

    if (fechaVen) {
      var fv = new Date(fechaVen);
      fv.setMinutes(fv.getMinutes() + fv.getTimezoneOffset());
      fv.setHours(0, 0, 0, 0);
      fechaVenField = { timestampValue: fv.toISOString() };
      if (hoy > fv) estado = 'VENCIDO';
    }

    writes.push({
      update: {
        name: dbPath + '/socios/' + dni,
        fields: {
          dni:              { stringValue: dni             },
          nombre:           { stringValue: nombreCompleto  },
          estado:           { stringValue: estado          },
          fechaVencimiento: fechaVenField,
          ultimaVisita:     { nullValue: null              }
        }
      }
    });
    total++;

    // Firestore acepta máximo 500 operaciones por batch
    if (writes.length === 500) {
      var r = UrlFetchApp.fetch(commitUrl, {
        method: 'POST', contentType: 'application/json',
        headers: { Authorization: 'Bearer ' + token },
        payload: JSON.stringify({ writes: writes }),
        muteHttpExceptions: true
      });
      if (r.getResponseCode() !== 200) throw new Error('Batch error: ' + r.getContentText());
      writes = [];
    }
  }

  if (writes.length > 0) {
    var res = UrlFetchApp.fetch(commitUrl, {
      method: 'POST', contentType: 'application/json',
      headers: { Authorization: 'Bearer ' + token },
      payload: JSON.stringify({ writes: writes }),
      muteHttpExceptions: true
    });
    if (res.getResponseCode() !== 200) throw new Error('Batch error: ' + res.getContentText());
  }

  throw new Error('OK - ' + total + ' socios sincronizados a Firestore');
}

  if (!hoja) { Logger.log('ERROR: No existe la hoja "Socios"'); return; }

  var datos = hoja.getDataRange().getValues();
  var token = getFirebaseToken_();
  if (!token) { Logger.log('ERROR: No se pudo obtener el token'); return; }

  var base = 'https://firestore.googleapis.com/v1/projects/new-life-app-f951c'
           + '/databases/(default)/documents/socios/';
  var maskParams = 'updateMask.fieldPaths=dni'
    + '&updateMask.fieldPaths=nombre'
    + '&updateMask.fieldPaths=estado'
    + '&updateMask.fieldPaths=fechaVencimiento';
  var hoy = new Date(); hoy.setHours(0, 0, 0, 0);
  var ok = 0; var err = 0;

  for (var i = 1; i < datos.length; i++) {
    var dni      = datos[i][0] ? String(datos[i][0]).trim() : '';
    var nombre   = datos[i][1] ? String(datos[i][1]).trim() : '';
    var apellido = datos[i][2] ? String(datos[i][2]).trim() : '';
    var fechaVen = datos[i][3];
    if (!dni) continue;

    var nombreCompleto = (nombre + ' ' + apellido).trim();
    var estado = 'AL DIA';
    var fechaVenField = { nullValue: null };

    if (fechaVen) {
      var fv = new Date(fechaVen);
      fv.setMinutes(fv.getMinutes() + fv.getTimezoneOffset());
      fv.setHours(0, 0, 0, 0);
      fechaVenField = { timestampValue: fv.toISOString() };
      if (hoy > fv) estado = 'VENCIDO';
    }

    var resp = UrlFetchApp.fetch(base + dni + '?' + maskParams, {
      method: 'PATCH',
      contentType: 'application/json',
      headers: { Authorization: 'Bearer ' + token },
      payload: JSON.stringify({
        fields: {
          dni:              { stringValue: dni            },
          nombre:           { stringValue: nombreCompleto },
          estado:           { stringValue: estado         },
          fechaVencimiento: fechaVenField
        }
      }),
      muteHttpExceptions: true
    });

    if (resp.getResponseCode() === 200) {
      ok++;
    } else {
      err++;
      Logger.log('Error en DNI ' + dni + ': ' + resp.getContentText());
    }
  }

  Logger.log('COMPLETADO: ' + ok + ' sincronizados, ' + err + ' errores');
}

// ── EDICIÓN DESDE LA APP ────────────────────────────────────────────────────
// Recibe POST con { originalDni, nombre, dni, fechaVencimiento } desde la app
// y actualiza la fila correspondiente en la hoja "Socios".
//
// Para activarlo (una sola vez):
//   Implementar → Nueva implementación → Aplicación web
//   Ejecutar como: Yo  |  Acceso: Cualquier usuario (anónimo) → Implementar
//   Copiá la URL y pegala en SHEETS_SYNC_URL de src/services/gymService.js
function doPost(e) {
  try {
    var data         = JSON.parse(e.postData.contents);
    var action       = String(data.action || 'update').trim();
    var ss           = SpreadsheetApp.getActiveSpreadsheet();
    var hoja         = ss.getSheetByName('Socios');
    if (!hoja) return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: 'Hoja no encontrada' }))
      .setMimeType(ContentService.MimeType.JSON);

    // ── AGREGAR nuevo socio ────────────────────────────────────────────────────
    if (action === 'add') {
      var addNombre   = String(data.nombre || '').trim();
      var addDni      = String(data.dni    || '').trim();
      var addFechaISO = data.fechaVencimiento || null;

      var partes   = addNombre.split(' ');
      var nombre   = partes[0] || '';
      var apellido = partes.slice(1).join(' ');
      var fechaVal = addFechaISO ? new Date(addFechaISO) : '';

      hoja.appendRow([addDni, nombre, apellido, fechaVal]);
      Logger.log('doPost add: ' + addDni + ' / ' + addNombre);
      return ContentService
        .createTextOutput(JSON.stringify({ ok: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── ELIMINAR socio ─────────────────────────────────────────────────────────
    if (action === 'delete') {
      var delDni  = String(data.dni || '').trim();
      var valores = hoja.getDataRange().getValues();
      var fila    = -1;
      for (var i = 1; i < valores.length; i++) {
        if (String(valores[i][0]).trim() === delDni) { fila = i + 1; break; }
      }
      if (fila === -1) return ContentService
        .createTextOutput(JSON.stringify({ ok: false, error: 'DNI no encontrado: ' + delDni }))
        .setMimeType(ContentService.MimeType.JSON);

      hoja.deleteRow(fila);
      Logger.log('doPost delete: ' + delDni);
      return ContentService
        .createTextOutput(JSON.stringify({ ok: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── ACTUALIZAR socio (comportamiento original) ─────────────────────────────
    var originalDni  = String(data.originalDni || '').trim();
    var nuevoNombre  = String(data.nombre       || '').trim();
    var nuevoDni     = String(data.dni          || '').trim();
    var fechaVencISO = data.fechaVencimiento    || null;

    var valores = hoja.getDataRange().getValues();
    var fila = -1;
    for (var i = 1; i < valores.length; i++) {
      if (String(valores[i][0]).trim() === originalDni) { fila = i + 1; break; }
    }
    if (fila === -1) return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: 'DNI no encontrado: ' + originalDni }))
      .setMimeType(ContentService.MimeType.JSON);

    var partes   = nuevoNombre.split(' ');
    var nombre   = partes[0] || '';
    var apellido = partes.slice(1).join(' ');

    hoja.getRange(fila, 1).setValue(nuevoDni);
    hoja.getRange(fila, 2).setValue(nombre);
    hoja.getRange(fila, 3).setValue(apellido);
    if (fechaVencISO) {
      hoja.getRange(fila, 4).setValue(new Date(fechaVencISO));
    } else {
      hoja.getRange(fila, 4).clearContent();
    }

    Logger.log('doPost update: ' + originalDni + ' → ' + nuevoDni + ' / ' + nuevoNombre);
    return ContentService
      .createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    Logger.log('doPost error: ' + err.message);
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
